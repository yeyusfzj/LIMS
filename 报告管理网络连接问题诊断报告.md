## 🎯 根本原因

**后端缺少报告管理的API路由实现!**

前端调用 `POST /api/v1/reports/generate`,但后端没有对应的路由处理器,导致返回 **404 Not Found** 或连接错误。

#### 2. 创建报告服务 (`backend-api/app/services/report_service.py`)

```python
"""报告服务"""

from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import select
from typing import Optional, List
from datetime import datetime

from app.models.report import Report, ReportTemplate
from app.models.sample import Sample
from app.core.exceptions import NotFoundException

class ReportService:
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def generate_report(
        self,
        sample_id: str,
        template_id: str,
        preview: bool = False,
        generated_by: str = None
    ) -> Report:
        """生成报告"""
        # 1. 获取样品信息
        sample = await self.get_sample(sample_id)
        if not sample:
            raise NotFoundException(f"样品不存在: {sample_id}")
        
        # 2. 获取模板信息
        template = await self.get_template(template_id)
        if not template:
            raise NotFoundException(f"模板不存在: {template_id}")
        
        # 3. 填充模板内容
        content = self.fill_template(template, sample)
        
        # 4. 生成报告编号
        report_number = await self.generate_report_number()
        
        # 5. 创建报告记录
        report = Report(
            reportNumber=report_number,
            sampleId=sample_id,
            templateId=template_id,
            content=content,
            status="DRAFT",
            generatedBy=generated_by,
            generatedAt=datetime.utcnow()
        )
        
        if not preview:
            self.db.add(report)
            await self.db.commit()
            await self.db.refresh(report)
        
        return report
    
    # ... 其他方法
```

```python
from pydantic import BaseModel, Field
from typing import Optional, List
from datetime import datetime
from enum import Enum

class ReportStatus(str, Enum):
    """报告状态枚举"""
    DRAFT = "DRAFT"
    PENDING_SIGNATURE = "PENDING_SIGNATURE"
    SIGNED = "SIGNED"
    DISTRIBUTED = "DISTRIBUTED"
    RECALLED = "RECALLED"

class ReportGenerate(BaseModel):
    """生成报告请求模型"""
    sampleId: str = Field(..., description="样品ID")
    templateId: str = Field(..., description="模板ID")
    preview: bool = Field(default=False, description="是否预览")

class ReportResponse(BaseModel):
    """报告响应模型"""
    id: str
    reportNumber: str
    sampleId: str
    templateId: str
    content: str
    status: ReportStatus
    generatedBy: str
    generatedAt: datetime
    
    model_config = {
        "from_attributes": True
    }
```
